cmake_minimum_required(VERSION 3.18)
project(canny_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(canny_scalar src/canny_scalar.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(canny_scalar PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(canny_scalar PRIVATE -fno-vectorize -fno-slp-vectorize)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(canny_scalar PRIVATE -fno-tree-vectorize -fno-tree-slp-vectorize)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_hysteresis src/test_hysteresis.cpp)
  add_test(NAME hysteresis_connectivity COMMAND test_hysteresis)
endif()
option(BUILD_OPENCV_CUDA "Build OpenCV CUDA reference (requires CUDA-enabled C++ OpenCV)" OFF)
if(BUILD_OPENCV_CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_package(OpenCV REQUIRED COMPONENTS core imgcodecs cudaimgproc cudafilters)
  add_executable(canny_opencv_cuda src/canny_opencv_cuda.cpp)
  target_include_directories(canny_opencv_cuda PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(canny_opencv_cuda PRIVATE ${OpenCV_LIBS} CUDA::cudart)
endif()
